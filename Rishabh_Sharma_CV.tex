\section{Research Experience}
\role{Simons Center Independent Postdoctoral Fellow}{Jul 2025 -- present}
{Simons Center for Computational Physical Chemistry, New York University}

\project{Cyclic Denoising Reveals Ultrastable Memories in Diffusion Models}
{\href{https://arxiv.org/abs/2606.24000}{Paper} \enspace\textbar\enspace
\href{https://rishabh-tifr.github.io/cyclic-denoising/movies/}{Landscape animations}}
{\textbf{NeurIPS 2026, Main Track}}
\begin{itemize}
\item \textbf{Invented cyclic denoising}, translating Ph.D. insights on memory and annealing in driven
disordered systems into an extraction attack and dynamical probe. Controlled forward/reverse diffusion
cycles reveal a \textbf{basin-hopping picture} of the generative landscape: memorized images occupy
ultrastable basins, persist for \textbf{thousands of cycles}, and regenerate after near-total corruption.
\item Efficiently \textbf{recovers memorized training images} even when training captions are unavailable,
a setting where caption-dependent extraction attacks cannot operate. Finds candidates in
\textbf{Stable Diffusion v1.4 and a pixel-space DDPM} through long-time dynamics with sampler-level
control, requiring no prompts, prior training-data knowledge, auxiliary
detectors, gradients, or weight inspection.
\end{itemize}

\project{Cross-View World Models}
{\href{https://arxiv.org/abs/2602.07277}{Paper} \enspace\textbar\enspace
\href{https://rishabh-tifr.github.io/cross-view-world-models/}{Project \& demos}}
{\href{https://openreview.net/forum?id=0g7De0sXWM}{\textbf{ICLR 2026 Workshop on World Models}}
}
\begin{itemize}
\item Led a collaboration with NYU neuroscientists and industry partner Statespace. Introduced an
objective to predict future observations from another viewpoint, training viewpoint- and
action-conditioned \textbf{diffusion transformers} on 4,186 one-minute Aimlabs episodes from four synchronized cameras.
\item Identified a \textbf{view-invariant ``cognitive map''} in middle transformer layers: linear pose probes
transfer between egocentric and bird's-eye views without retraining. Yaw interventions rigidly rotate
its heading ring by the commanded angle, leaving earlier, view-dependent heading unchanged.
\end{itemize}

\vspace{4pt}
\role{Doctoral Researcher}{2019 -- 2025}
{Tata Institute of Fundamental Research (TIFR), Hyderabad, India}
\begin{itemize}
\item Discovered a correspondence between activity-induced and cyclic-shear annealing in amorphous
solids, linking annealing to a ductile-to-brittle transition
(\href{https://doi.org/10.1038/s41567-024-02724-5}{\textbf{\textit{Nature Physics}}}, 2025;
featured in \href{https://www.nature.com/articles/s41567-025-02900-1}{\textit{News \& Views}}).
\item Developed fast, fault-tolerant mechanical-memory protocols
(\href{https://doi.org/10.1103/PhysRevLett.134.018202}{\textbf{\textit{Physical Review Letters}}},
2025; co-first author). Conducted large-scale molecular-dynamics simulations using \textbf{C and MPI}
on HPC clusters.
\end{itemize}
